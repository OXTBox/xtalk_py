function T.HeightAt x, z
   put sGridN * sGridS / 2 into sHalf
   repeat with i = 1 to 100
      put (i * 7) mod 23 into sH[i]
   end repeat
   return Hunter.HeightAt(x, z)
end T.HeightAt

function T.RandomGemOK
   -- RandomGem is random: check 200 picks are all one of the six gem names
   put "Diamond,Sapphire,Ruby,Emerald,Amethyst,Topaz" into tNames
   repeat 200 times
      put Hunter.RandomGem() into tGem
      if tGem is not among the items of tNames then return "bad:" & tGem
   end repeat
   return "ok"
end T.RandomGemOK
