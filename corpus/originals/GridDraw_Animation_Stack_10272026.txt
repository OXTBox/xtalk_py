function binaryStringToPBM pBinaryString
   put the number of chars of line 1 of pBinaryString into tWidth
   put the number of lines of pBinaryString into tHeight
   
   put "P1" & cr into tPBM
   put tWidth && tHeight & cr after tPBM
   
   repeat with y = 1 to tHeight
      put line y of pBinaryString & cr after tPBM
   end repeat
   
   return tPBM
end binaryStringToPBM

function createBlankPBM pGridSize
   put "P1" & cr into tPBM
   put pGridSize && pGridSize & cr after tPBM
   repeat pGridSize
      put repeatChar("0",pGridSize) & cr after tPBM
   end repeat
   return tPBM
end createBlankPBM

function repeatChar pChar,pCount
   put empty into tResult
   repeat pCount
      put pChar after tResult
   end repeat
   return tResult
end repeatChar

on copyFrame tkey
   saveForUndo
   if bEditFont is true then SaveEditedCharacter 
   switch tkey
      case "left"
         put gCurrentCharacterImage-1 into gCurrentCharacterImage
         break
      case "right"
         put gCurrentCharacterImage+1 into gCurrentCharacterImage
         break
   end switch
   --put "copyframe"
   --if "FontViewer" is among the lines of the windows then updateFontViewerPreview gCurrentCharacterImage
   set the customPropertySet of stack (the mainStack of this stack) to FontData
   do "put the" && gCurrentFont && "of stack (the mainStack of this stack) into tFontArray"
   put gCurrentCharacterImage-1 into gCurrentAsciiVal
   put numtochar(gCurrentAsciiVal) into gCurrentCharacter 
   put buildbitwise(gBinaryString) into tFontArray[gCurrentAsciiVal]
   do "set the" && gCurrentFont && "of stack (the mainStack of this stack) to tFontArray"
   set the customPropertySet of stack (the mainStack of this stack) to empty
   DrawFromBitFontdata gCurrentAsciiVal,gCurrentFont 
   if bEditFont is true then SaveEditedCharacter 
   if "FontViewer" is among the lines of the windows then updateFontViewerPreview gCurrentCharacterImage
end copyFrame

-- Diagonal step
on twostepshift tDirection
   lock screen
   put abs(tDirection)-gGridSize into tremainder
   if tDirection <0 then 
      shiftPixels -gGridSize
      shiftPixels tRemainder*-1
   else
      shiftPixels gGridSize
      shiftPixels tRemainder
   end if
   unlock screen
end twostepshift
-- move filled pixels right, left, up and down
on shiftPixels tDirection
   if tDirection is an integer then
      if abs(tDirection) > gGridSize then
         twostepshift tDirection
         exit shiftPixels
      end if
   end if
   --put GridToBinaryBox() INTO gBinaryString
   saveForUndo
   put "" into tNewBinaryString
   put number of lines in gBinaryString  into NLines
   switch tDirection
      case "left"
      case -1
         repeat with i = 1 to Nlines
            put char 2 to -1 of line i of  gBinaryString  &char 1 of line i of  gBinaryString & cr after tNewBinaryString
         end repeat
         break
      case "right"
      case 1
         repeat with i = 1 to Nlines
            put char -1 of line i of  gBinaryString  &char 1 to -2   of line i of  gBinaryString & cr after tNewBinaryString
         end repeat
         
         break
      case "up"
      case -gGridSize
         put line 2 to -1 of gBinaryString & cr & line 1 of gBinaryString into tNewBinaryString
         
         break
      case "down"
      case gGridSize
         put  line -1 of gBinaryString & cr & line 1 to -2 of gBinaryString into tNewBinaryString
         break
   end switch
   put char  1 to -2 of tNewBinaryString into gBinaryString 
   saveForUndo
   UpDateGridFromBinaryString
   --return tNewBinaryString
end shiftPixels

on flipPixels tDirection
   -- refresh the binary string from graphics backgrounds
   --put GridToBinaryBox() INTO gBinaryString
   saveForUndo
   put "" into tNewBinaryString
   put number of lines in gBinaryString  into NLines
   lock screen
   switch tDirection
      case "up" -- flip horizontal
      case "down"
         repeat with i = gGridSize down to 1
            put line i of gBinaryString & cr after  tNewBinaryString
         end repeat
         break
      case "left"
         lock screen
         flipPixels "rot-90"
         flipPixels "rot-90"
         flipPixels "up"
         unlock screen
         break
      case "right"
      case "horizontal"
         flipPixels "rot-90"
         flipPixels "rot-90"
         flipPixels "down"
         break
      case "rot-90" -- rotate -90
         repeat with i = gGridSize down to 1
            put "" into tLine
            repeat with j = 1 to gGridSize
               put char i of line j of gBinaryString after tLine
            end repeat
            put tLine & cr after  tNewBinaryString
         end repeat
         
         break
      case "rot90" -- rotate 90
         repeat with i = 1  to gGridSize
            put "" into tLine
            repeat with j = gGridSize down to 1
               put char i of line j of gBinaryString after tLine
            end repeat
            put tLine & cr after  tNewBinaryString
         end repeat
         break
   end switch
   put char  1 to -2 of tNewBinaryString into gBinaryString 
   UpDateGridFromBinaryString
   unlock screen
end flipPixels

function addallchars tString
   replace cr with empty in tString
   put 0 into tTotal
   repeat for each char  I in tString
      if I is a number then add I to tTotal
   end repeat
   return tTotal
end addallchars

function addAllItems tItemList,tDelimiter
   set itemdelimiter to tDelimiter
   put 0 into tTotal
   repeat for each item I in tItemList
      if I is a number then add I to tTotal
   end repeat
   return tTotal
end addAllItems

on ImportImage tFileName
   if tFileName is empty then
      answer file "Import Image PNG"
      if it is empty then exit ImportImage
      put it into tFileName
   end if
   lock screen
   put gGridSize & "x" & gGridSize into tSizeFolder
   if not exists(image "16x16") then  create image "16x16"
   set the dontDither of image "16x16" to true
   set the fileName of image "16x16" to empty
   --wait 30 milliseconds
   set the fileName of image "16x16" to tFileName
   --wait 200 milliseconds
   flip image "16x16" vertical  -- the data is flipped when read back in?
   set the width of  image "16x16" to 16
   set the height of  image "16x16" to 16
   set the BOTTOMLEFT of image "16x16" to -1,-1
   put the imageData of image "16x16" into tData
   init.template.import.image "import",16,tData
   set the layer of image "16x16" to BOTTOM
   hide image "16x16"
   
   --get the binary string of image data
   put getImageBinaryColor(16,16,"16x16",-1,-1)  into gBinaryString
   -- sometimes the data is all black for some reason, retry import.
   if addAllItems(gBinaryString) =0 and tImportAttempted <1 then
      put getImageBinaryColor(16,16,"16x16",0,0)  into gBinaryString
      if addAllItems(gBinaryString) <1  then
         add 1 to tImportAttempted
         send ImportImage && tFileName to this stack in 100 milliseconds
         exit ImportImage
      end if
   end if
   -- reset the binary grid and set all the graphics to white
   --draw from  binary string
   put 0 into tImportAttempted
   UpdateGridFromBinaryString
   unlock screen
   set the lastcontrolKey of stack (the mainStack of this stack) to empty 
end ImportImage

on ImportMapImage tFileName
   if tFileName is empty then
      answer file "Import Image PNG"
      if it is empty then exit ImportMapImage
      put it into tFileName
   end if
   lock screen
   if not exists(image "MapImage") then  create image "MapImage"
   set the dontDither of image "MapImage" to true
   set the fileName of image "MapImage" to empty
   set the fileName of image "MapImage" to tFileName
   wait 100 milliseconds
   flip image "MapImage" vertical  -- the data is flipped when read back in?
   set the width of  image "MapImage" to 512
   set the height of  image "MapImage" to 512
   set the BOTTOMLEFT of image "MapImage" to -1,-1
   put the imageData of image "MapImage" into tData
   set the layer of image "MapImage" to BOTTOM
   hide image "MapImage"
   unlock screen
   --get the binary string of image data
   repeat with i = 1 to 512 step gGridSize
      repeat with j = 1 to 512 step gGridSize
         put getOffsetImageBinaryColor(i,j,gGridSize,gGridSize,tData)  into gBinaryString
         
      end repeat
   end repeat
   
   -- sometimes the data is all black for some reason, retry import.
   if addAllItems(gBinaryString) <3 and tImportAttempted =0 then
      add 1 to tImportAttempted
      
      --send ImportMapImage && tFileName to this stack in 100 milliseconds
      exit ImportMapImage
   end if
   -- reset the binary grid and set all the graphics to white
   --draw from  binary string
   UpdateGridFromBinaryString
   set the lastcontrolKey of stack (the mainStack of this stack) to empty 
end ImportMapImage

Function emptyBinaryMatrix tDepth,tWidth
   put 0 into n
   repeat with i = 1 to (tDepth*tWidth)
      add 1 to n
      put 0 after tOUt
      if n= tWidth then
         put 0 into n
         put cr after tOut
      end if
   end repeat
   return char 1 to -1 of tOut
end emptyBinaryMatrix


-- pImg is the name of the img,
-- x is the col of the pixel, y the row of the pixel
function getPixelRGB pImg,x,y
   put the width of img pImg into w
   put 4*((y-1)*w+x) into k
   put byte k-2 to k of the imageData of img pImg into b
   return (byteToNum(byte 1 of b),byteToNum(byte 2 of b), byteToNum(byte 3 of b))
end getPixelRGB

function getPixelRGBA pImg,x,y -- return R,G,B,A
   
   put the width of img pImg into w
   put 4*((y-1)*w+x) into k
   put byte k-2 to k of the imageData of img pImg into b
   put byteToNum(byte (y-1)*w+x of the alphaData of img pImg) into c
   return (byteToNum(byte 1 of b),byteToNum(byte 2 of b), byteToNum(byte 3 of b), c) 
end getPixelRGBA

function getImageBinaryColor pWidth,pHeight,pImage
   set itemDel to comma
   repeat with y = 1 to pHeight
      put "" into tBitmapLine -- Initialize the line string
      repeat with x =  1 to pWidth
         put getPixelRGBA("16x16",x,y) into tRGBA
         -- if there is no alpha then item 4 would be 255 on every line
         put addallLines(4,tRGBA) into allrgba
         if addallLines(4,tRGBA) < (the number of lines of tRGBA)*255 then
            -- imports transparency
            if tRGBA is "0,0,0,0" then
               put "0"  after tBitmapLine -- White pixel
            else
               put "1" after tBitmapLine -- Non-white pixel
            end if
         else
            -- imports flat image
            if tRGBA is "255,255,255,255" then
               put "0"  after tBitmapLine -- White pixel
            else
               put "1" after tBitmapLine -- Non-white pixel
            end if
         end if
      end repeat
      put tBitmapLine & cr after tBitmap
   end repeat
   --put allRGBA
   return tBitmap
end getImageBinaryColor

function getOffsetImageBinaryColor tx,ty,pWidth,pHeight,pImage
   --tx,ty are starting pixels in an image
   set itemDel to comma
   repeat with y = ty to ty+pHeight
      put "" into tBitmapLine -- Initialize the line string
      repeat with x =  tx to tx+pWidth
         put getPixelRGBA("MapImage",x,y) into tRGBA
         -- if there is no alpha then item 4 would be 255 on every line
         put addallLines(4,tRGBA) into allrgba
         if addallLines(4,tRGBA) < (the number of lines of tRGBA)*255 then
            -- imports transparency
            if tRGBA is "0,0,0,0" then
               put "0"  after tBitmapLine -- White pixel
            else
               put "1" after tBitmapLine -- Non-white pixel
            end if
         else
            -- imports flat image
            if tRGBA is "255,255,255,255" then
               put "0"  after tBitmapLine -- White pixel
            else
               put "1" after tBitmapLine -- Non-white pixel
            end if
         end if
      end repeat
      put tBitmapLine & cr after tBitmap
   end repeat
   --put allRGBA
   return tBitmap
end getOffsetImageBinaryColor

on UnpackBinaries
   set the customPropertySet of stack (the mainStack of this stack) to Binaries
   put the customKeys of  stack (the mainStack of this stack) into tBinariesKeys
   put the number of lines of tBinariesKeys into tMaxLines
   put the SavedBinary of  stack (the mainStack of this stack)into tBigArray

   sort tBigArrayKeys Numeric Ascending
   put the number of lines of tBigArrayKeys into tEndValue
   if tEndValue < 255 then exit UnpackBinaries
   put tMaxLines into tCount
   repeat with i = 1 to number of lines of tBigArrayKeys  step 255
      put 0 into n

      put "SavedBinary_"&tCount into tBinaryTable

         put tBigArray[K] into tNewArray[n]
         add 1 to n
         do "set the" && tBinaryTable && "of  stack (the mainStack of this stack) to tNewArray"
      end repeat
      add 1 to tCount
   end repeat
   set the customPropertySet of stack (the mainStack of this stack) to empty
   EmptySavedBinary

-- retrieve 8x8 pixel bit data strings, return with or without padding for 16x16
function getEightBit tNumber,bPadded,pSize
   if tNumber is not a number then put chartonum(char 1 of tNumber) into tNumber
   put getCProp("Eightbit","Aquarius") into t8BitBinaryString
   put line tNumber of t8BitBinaryString into tLine
   
   put widestringToGrid(tLine,8) into tBlock
   if bPadded is true then
      if pSize is empty then put 16 into pSize
      put stringRepeat(pSize,"0") into tEmptySpaceLines
      put stringRepeat((pSize/4),"0") into tEmptySpaceColumns
      put stringRepeat((pSize/4),tEmptySpaceLines&cr) into tEmptySpaceChunk
      repeat for each line L in tBlock
         put tEmptySpaceColumns & L & tEmptySpaceColumns &cr after tPaddedOutput
      end repeat
      put tEmptySpaceChunk & tPaddedOutput & tEmptySpaceChunk into finalOutput
   else
      return tBlock
   end if
   return finalOutput
end getEightBit

end widestringToGrid

function UnpadNumber tNum
   repeat with i = the number of chars of tNum down to 1
      if char 1 of tNum is "0" then put char 2 to -1 of tNum into tNum
   end repeat
   return tNum
end UnpadNumber

// bitfontmaker2 data processing bitwise operation find out which bit in row is lit
function getFilledColumns inputValue
   put "1,2,4,8,16,32,64,128,256,512,1024,2048,4096,8192,16384,32768" into columnValues
   set itemDel to comma
   put empty into filledColumns
   repeat with i = number of items in columnValues down to 1
      if inputValue >= item i of columnValues then
         put i & comma after filledColumns
         
         subtract item i of columnValues from inputValue
      end if
   end repeat
   delete the last char of filledColumns -- remove trailing comma
   return filledColumns
end getFilledColumns

-- bitwise operation turns long 010101 style binary to compressed bitwise binary for BitFontMaker2 (and cleaner data)
Function BuildBitWise pBinaryString
   if pBinaryString is empty then put gBinaryString into pBinaryString
   put pBinaryString into gBinaryString
   put "1,2,4,8,16,32,64,128,256,512,1024,2048,4096,8192,16384,32768" into columnValues
   put gBinaryString into tBits
   set itemDel to comma
   repeat with i = 1 to gGridSize
      put 0 into tLineTotal
      repeat with j = 1 to gGridSize
         if char j of line i of tBits is not 0 then add item j of columnValues to  tLineTotal
      end repeat
      put tLineTotal & comma after AllTotals
   end repeat
   --put AllTotals
   return char 1 to -2 of AllTotals
end BuildBitWise

function  parseFrameInput tFrames
   replace comma with cr in tFrames
   repeat with F = 1 to number of lines of tFrames
      put line F of tFrames  into tCheckItem
      set itemDel to "-" 
      if number of items of tCheckItem > 1 then
         repeat with n=item 1 of tCheckItem to item 2 of tCheckItem
            put n & comma after tNewItems 
         end repeat
         put char 1 to -2 of tNewItems into line F of tFrames
      end if
   end repeat
   replace cr with comma in tFrames
   return tFrames
end parseFrameInput

on shiftFontPosition tDistance
   put 0 into tChar
   put "" into tNewArray
   set the customPropertySet of stack (the mainStack of this stack) to FontData
   do "put the"&& gCurrentFont && "of stack (the mainStack of this stack)into tAsciiArray"
   put the keys of tAsciiArray into tKeys
   filter lines of tKeys without empty
   sort lines of tKeys numeric ascending
   put empty into tLoopAround
   if tDistance > 0 then 
      put tDistance*-1 into tEnd
      put line 1 to (tEnd-1) of tKeys into  tStepsBack 
      if number of lines of tStepsBack < number of lines of tKeys then
         put line tEnd to -1 of tKeys into tWrapAround
         put cr & tStepsBack after tWrapAround
      end if
      put abs(tDistance) into tChar
   else 
      put tDistance into tEnd
      put abs(tDistance) into tStart
      put line 1 to tStart of tKeys into tWrapAround
      if number of lines of tWrapAround < number of lines of tKeys then
         put line tStart+1 to -1 of tKeys &cr before tWrapAround
      end if
      put tDistance into tChar
   end if
   
   repeat for each line K in tWrapAround
      put tAsciiArray[k] into tNewArray[tChar]
      put gcurrentFOntFolder&"/"&k&".png" into tFileName
      put gcurrentFOntFolder&"/"&tChar&".png" into tNewFileName
      rename tFileName to tNewFileName
      add 1 to tChar
      if tChar > 255 then put 0 into tChar
   end repeat
   filter lines of tWrapAround without empty
   
   do "set the"&& gCurrentFont && "of stack (the mainStack of this stack)to tNewArray"
   set the customPropertySet of stack (the mainStack of this stack) to empty
   send showCurrentFont to stack (the mainStack of this stack)  in 300 milliseconds
end shiftFontPosition

   put lineOffset(tEndtext,tStackScript) into tEnd

   put line tStart to tEnd of tStackScript into tCommandText

-- Function to check if a point is within a rectangle
function pointInRectangle pX, pY, pRect
   repeat with i = 1 to the number of lines in pRect
      put line i of pRect into tLine
      repeat with j = 1 to the number of chars in tLine
         if charToNum(char j of tLine) = 1 then
            put (pX + j - 1) & "," & (pY + i - 1) & return after tPoints
         end if
      end repeat
   end repeat
   return tPoints
end pointInRectangle

-- Function to check if a rectangle overlaps with existing rectangles
function rectangleOverlaps pRect, pRectangles
   repeat with i = 1 to the number of lines in pRectangles
      put line i of pRectangles into tExistingRect
      put item 1 of pRect & comma & item 2 of pRect into tTopLeft
      put item 3 of pRect & comma & item 4 of pRect into tBottomRight
      
      if item 1 of pRect is empty then exit rectangleOverlaps
      repeat with x = item 1 of Prect to item 3 of Prect
         if (x,item 2 of pRect) or (x,item 4 of pRect) is within tExistingRect then return true
      end repeat
      repeat with y = item 2 of Prect to item 4 of Prect
         if (item 1 of prect,y) or (item 3 of prect,y) is within tExistingRect then return true
      end repeat
   end repeat
   return false
end rectangleOverlaps

-- Function to generate a random rectangle
function generateRectangle 
   put "" into tRect
   put random(4) + 3 into tWidth
   put random(4) + 3 into tHeight
   put random(cGridWidth - tWidth + 1) into tX
   put random(cGridHeight - tHeight + 1) into tY
   if tx > gGridSize then put gGridSize into tx
   if ty > gGridSize then put gGridSize into ty
   return tX & "," & tY &","&tx+tWidth &","&ty+tHeight     --& return & tRect
end generateRectangle

--it generates a binary string representing white and non-white pixels.

--Adjust the pGridWidth and pGridHeight parameters as needed.

function dissectImageIntoGrids pGridWidth, pGridHeight, pImage

   put 0 into gCurrentAsciiVal
   repeat with startY = 1 to tImageHeight step pGridHeight
      do "put the"&&gCurrentFont&& "of stack (The mainStack of this stack) into tFontArray"
      repeat with startX = 1 to gImageWidth step pGridWidth
         put getImageBinaryColor(pGridWidth, pGridHeight, pImage, startX, startY) into tBinaryString

         put tBinaryString & return after gBinaryStrings
      end repeat
   end repeat
   set the custompropertyset of stack (The mainStack of this stack) to empty
   delete image pimage
   return gBinaryStrings
end dissectImageIntoGrids
