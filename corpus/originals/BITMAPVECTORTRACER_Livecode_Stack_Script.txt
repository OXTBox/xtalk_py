function MouseLoc.To_Column_Row gridWith,gridHeight,maxColums,maxRows
   
   --- Get  Column and Row of the grid that the mouse was clicked in
   put trunc((item 1 of the mouseLoc) / gridWidth ) into columnNumber
   put trunc((item 2 of the mouseLoc) / gridHeight )  into rowNumber
   
   --- Mouse outside window doesn't get counted as a bit!
   if columnNumber <1  then put 1 into columnNumber
   if columnNumber > maxColums then put maxColums into columnNumber
   if rowNumber <1  then put 1 into rowNumber
   if rowNumber > maxRows then put maxRows into rowNumber
   return columnNumber,rowNumber
end MouseLoc.To_Column_Row

function Define.EmptyImagePBMData
   repeat with i = 1 to gGridColumns
      put "0"  after tLine
   end repeat
   repeat with i = 1 to gGridRows
      put tLine & cr after tData
   end repeat
   filter lines of tData without empty
   put line 1 to -1 of tData into gImagePBMBinaryData
   return gImagePBMBinaryData
end Define.EmptyImagePBMData


function Prop.getBinaryHistory n
   put the BinaryHistory of this card into tBinaryArray
   put the number of lines of the keys of tBinaryArray+1 into tCount
   put the keys of tBinaryArray into binKeys
   if tCount <1 then return empty
   if n is empty or n > tCount then put tCount into n
   return tBinaryArray["bit_"&padNumber(n,4)]
end Prop.getBinaryHistory

function string.isEmpty tString,tFilter
   filter tString without tFilter
   filter tString without empty
   return (tString is empty)
end string.isEmpty




on BinaryDataToPBMImage tBinaryString
   lock screen
   --- create an string of 32x32 zeros if there is no data availale
   if tBinaryString is empty then 
      put Define.EmptyImagePBMData() into gImagePBMBinaryData
      put gImagePBMBinaryData into tBinaryString
   end if
   
   --- pass data  to the global
   put tBinaryString into gImagePBMBinaryData
   --- save the changes to history property
   if bPlaying is false then  Prop.appendBinaryHistory gImagePBMBinaryData
   
   put gImagePBMBinaryData into pImageData
   
   put "" into pImageData
   -- pad the characters in the data with spaces to create the PBM data
   repeat for each char tBit in tBinaryString
      put tBit & space after pImageData
   end repeat
   
   --- truncate any extra line that might have been added by editing
   put line 1 to gGridRows of pImageData into pImageData
   
   put the number of lines of pImageData into tVertLength 
   
   --- iterate over the data, is it just returning the data we had already?
   repeat with y = 1 to tVertLength
      put the number of words of line y of pImageData into tLineHorzLength
      repeat with x = 1 to tLineHorzLength
         get word x of line y of pImageData
         if it = 0 or it = 1 then 
            put it &" " after tNewLine
         else 
            --put it & cr after msg
            next repeat
         end if
      end repeat
      put tNewLine & cr after tData
      put empty into tNewLine
   end repeat
   
   filter lines of tData without empty
   put "P1" & cr & the number of words of line 1 of tData && the number of lines of tData & cr & tData into tData
   
   if there is not an image "PBM_IMAGE" then create image "PBM_IMAGE"
   set the text of image "PBM_IMAGE" to empty
   set the text of image "PBM_IMAGE" to tData
   set the alphaData of image "PBM_IMAGE" to empty
   set the height of image "PBM_IMAGE" to width of this card/2    ---tVertLength*gGridWidth
   set the width of image "PBM_IMAGE" to height of this card/2   ---tLineHorzLength*gGridWidth
   
   set the topLeft of image "PBM_IMAGE" to 0,0
   unlock screen
end BinaryDataToPBMImage

function math.GridScale tMin
   
   put the number of graphics of this card into tnumGraphics
   put the originalwidth of grc 1 of this card into tOriginalWidth
   if tOriginalWidth = 0 or tOriginalWidth = empty  then return 1
   put the width of this card into tWidth
   --if tNumGraphics < tMin then
   put tMin into tNumGraphics
   put tWidth/tNumGraphics into tMinimalWidth
   
   if tOriginalWidth = 0 or tMinimalWidth = 0 then return 1
   put tMinimalWidth/tOriginalWidth into tAspectRatio
   put tAspectRatio
   return tAspectRatio
end math.GridScale

/*Exactly! You’ve got it right. Let’s break it down further:

Boundary Walls (Outside): 
The initial accumulation in WestWalls corresponds to the outer walls of the entire grid or map. These are the walls that enclose the entire area, forming the boundary.

Grid Unit Closures (Columns and Rows): 
As we traverse through the columns and rows of the grid, WestWalls accumulates information about the western walls of each individual grid unit. It keeps track of whether there’s a wall to the west of each unit.

When moving from left to right (column-wise), WestWalls records the presence or absence of walls to the west of each grid unit.
Similarly, when moving from top to bottom (row-wise), it continues to accumulate information about the western walls of each unit.
In summary, WestWalls helps keep track of the western walls as we explore the grid, ensuring we know which walls are present and where the boundaries lie*/

-- this turns grid x,y coordinates such as {10,30} into a 3D dungeon.
function textToMap tBinaryString
   ---  tBinaryString = grid of 1s and 0s
   put empty into textArray	--Array{} -- whole map as one long string of 1s and 0s each element of array = map, or an ascii character whatever you paint
   put empty into quadwalls	--Array{} -- the final "3D" defintion of where "quads" will be drawn
   put empty into QuadFloors  --- array of floor locations in the grid
   put "" into  WestWalls  -- WestWalls
   
   
   repeat with Y = 1 to number of lines of tBinaryString
      put line Y of tBinaryString into scanColumn
      
      repeat with  C =1 to number of chars of scanColumn 
         --floor
         put char C of scanColumn into QFloor
         put char C+1 of scanColumn into QEast
         put char C-1 of scanColumn into QWest
         
         --         put c & comma & y & cr after xystring -- string tracking all floors?
         put WestWalls & QWest into WestWalls  ---Tracks the WEST, Left hand  walls as we progress
         put QLfoor into textArray[number of lines of the keys of textArray+1] --- just tracking all positions?
         
         if QFloor <> "0" then
            put c,y into QuadFloors[number of lines of the keys of QuadFloors+1]
            put c , y & cr after xystring -- string tracking Walls?
            
            
            -- using quadwalls for roof and floor
            put c,y,"floor" into quadwalls[number of lines of the keys of quadwalls+1]
            put c,y,"roof" into quadwalls[number of lines of the keys of quadwalls+1]
            
            put QFloor into QRoof 
            if i <=  the number of lines of tBinaryString then 
               put char c of line Y+1 of tBinaryString into QSouth
               
               if QSouth <> "1" then put c,y+1,"north" into quadwalls[number of lines of the keys of quadWalls+1] 
            end if
            
            if i >= 2 then
               put char c of line Y-1 of tBinaryString into QNorth -- wall faces south
               if QNorth <> "1" then put c,y-1,"south" into quadwalls[number of lines of the keys of quadWalls+1] 
            end if
            --
            -- east and west
            if c < number of chars of scanColumn and QEast <> "1" then put c,y,"east" into quadwalls[number of lines of the keys of quadWalls+1] 
            
            --West
            if c > 1 and QWest <> "1" then put c,y,"west" into quadwalls[number of lines of the keys of quadWalls+1] 
            
            -- Border
            if y = 1 then    put c,y,"north" into quadwalls[number of lines of the keys of quadWalls+1] 
            
            if c = 1 then  put c,y,"west" into quadwalls[number of lines of the keys of quadWalls+1] 
            
            if c = number of chars of scanColumn then put c,y,"east" into quadwalls[number of lines of the keys of quadWalls+1] 
            
            if i = the number of lines of tBinaryString then put c,y,"south" into quadwalls[number of lines of the keys of quadWalls+1] 
            
         end if-- if
      end repeat-- columns
      
   end repeat -- y+1 into y
   set the Walls3D of this card to quadWalls
   set the FloorsAndCeilings of this card to QuadFloors
   return xystring
end  textToMap

/*This function checks if a given point (x, y) is an edge point in a 2D grid represented by the string tXYString. 
An edge point is defined as a point that has less than 8 neighbors in the grid. 
The function loops over the 3x3 neighborhood around the point (x, y), counts the number of valid neighbors, and returns whether this count is less than 8. 
The function excludes the center point (x, y) itself when counting the neighbors.*/

function isEdgePoint x, y, tEdge,tBinaryString
   put empty into tNeighbor  # Initialize the variable tNeighbor
   put empty into tNeighborCount  # Initialize the variable tNeighborCount
   repeat with dx = -1 to 1  # Loop over the x-axis from -1 to 1
      repeat with dy = -1 to 1  # Loop over the y-axis from -1 to 1
         if dx <> 0 or dy <> 0 then  # Exclude the center point
            put char (x+dx) of line (y+dy) of tBinaryString into tNeighbor  # Get the character at the neighboring position
            if tNeighbor is "0"  then  # Check if the neighboring character is in the string
               add 1 to tNeighborCount  # If yes, increment the count
            end if
         end if
      end repeat
   end repeat
   return tNeighborCount < tEdge  # Return true if the count is less than 8, indicating it's an edge point
end isEdgePoint


function Binary.GetFilled tBinaryString
   repeat with Y = 1 to number of lines of tBinaryString
      put line Y of tBinaryString into scanColumn
      repeat with  C =1 to number of chars of scanColumn 
         put isEdgePoint(C, y, 1, tBinaryString) into tBool
         if tBool is true then 
            put 0 after tResults
         else
            put "1" after tResults
         end if
      end repeat
      put cr after tResults
   end repeat
   return tResults
end Binary.GetFilled

function Binary.GetFieldFilledDynamic tEdge,tBinaryString

      put line Y of tBinaryString into scanColumn
      repeat with  C =1 to number of chars of scanColumn 
         put isEdgePoint(C, y, tEdge, tBinaryString) into tBool
         if tBool is true then 
            put 0 after tResults
         else
            put "1" after tResults
         end if
      end repeat
      put cr after tResults
   end repeat

function rectGridFromPoint tBinaryString,tWidth
   if tWidth is empty then put 20 into tWidth
   filter lines of tBinaryString without empty
   if tBinaryString is empty then return empty
   
   put textToMap(tBinaryString) into tXYString
   repeat for each line L in tXYString
      if L is empty then next Repeat
      put item 1 of L-1 into x
      put item 2 of L-1 into y
      if x<1 then put 1 into x
      if y<1 then put 1 into y
      
      -- convert bit to rectangle
      put rectFromPoint(x*tWidth,y*tWidth,tWidth) into tRect
      
      --- vertice  for every corner of rectangle
      put item 1 of tRect & comma & item 2 of tRect into TL
      put item 3 of tRect & comma & item 2 of tRect into TR
      put item 3 of tRect & comma & item 4 of tRect into BR
      put item 1 of tRect & comma & item 4 of tRect into BL
      
      ---  vertices to rectangle defintiion
      put TL & cr & TR & cr & BR & cr & BL & cr & TL into tFinalRect
      
      --- append rectangle to graphic point list
      put tFinalrect & cr & cr after rectGrid
   end repeat
   --- close up the last grid point
   put tFinalRect after rectGrid
   return rectGrid
end rectGridFromPoint

function rectGridFromPoint.ExcludeNeighbors tBinaryString,tWidth
   if tWidth is empty then put 20 into tWidth
   filter lines of tBinaryString without empty
   if tBinaryString is empty then return empty
   put textToMap.ExcludeNeighbors(tBinaryString) into tXYString
   repeat for each line L in tXYString
      if L is empty then next Repeat
      put item 1 of L into x
      put item 2 of L into y
      
      -- convert bit to rectangle
      put rectFromPoint(x*tWidth,y*tWidth,tWidth) into tRect
      
      --- vertice  for every corner of rectangle
      put item 1 of tRect & comma & item 2 of tRect into TL
      put item 3 of tRect & comma & item 2 of tRect into TR
      put item 3 of tRect & comma & item 4 of tRect into BR
      put item 1 of tRect & comma & item 4 of tRect into BL
      
      ---  vertices to rectangle defintiion
      put TL & cr & TR & cr & BR & cr & BL & cr & TL into tFinalRect
      
      --- append rectangle to graphic point list
      put tFinalrect & cr & cr after rectGrid
   end repeat
   return rectGrid
end rectGridFromPoint.ExcludeNeighbors

function Digitize.DeresolveHorizontal tStep,tSide
   if tStep is  empty then put the Seqeuence_step of this card into tStep
   add 1 to tStep
   set the Sequence_step of this card to tStep
   
   put the currentGrid of this Card into tBinaryString
   
   put Binary.GetFieldFilledDynamic(tStep,tBinaryString) into tExclusionGrid
   --put tExclusionGrid into fld "TheGrid"
   repeat with y = 0 to number of lines of tBinaryString
      put  line y of tBinaryString into scanLine
      put  line y of tExclusionGrid into tKnockouts
      switch tSide  
         case "left"
            repeat with x = 1 to number of chars of scanLine
               if char x of tKnockouts = "0" and char x+tStep of tKnockouts <> "1" then put char x of tKnockOuts into char x of scanLine
            end repeat
            break
         case "right"
            repeat with x = 1 to number of chars of scanLine
               if char x of tKnockouts = "0" and char x+tStep of tKnockouts <> "0" then put char x of tKnockOuts into char x of scanLine
            end repeat
            break
      end switch 
      put scanLine & cr after tScannedInput
   end repeat
   filter lines of tScannedInput without empty
   --put tScannedInput into fld "exclusions"
   return tScannedInput
end Digitize.DeresolveHorizontal

on translate.luaToaDegree
   put field "Lua" into tLua
   replace tab with empty in tLua
   replace space & space with space in tLua
   put tLua into fld "lua"
   set the OGLua of me to tLua
   replace cr & space with cr in tLua
   -- replace x = 1 with put 1 into x
   repeat with i = 1 to number of lines of tLua
      put line i of tLua into L
      set itemDel to "="
      if "=" is among the chars of L then
         if word 1 of item 1 of L is not "if" then
            if word 1 of item 2 of L = "{}" then put "empty "&"-"&"-Array" before  of item 2 of L
            put "put" && word 1 of item 2  of L && "into" && item 1 of L  into tFix
            
            if number of words  of item 2  of L > 1 then  put tab  & word 2  to -1 of L after tFix
            put tFix into line i of tLua
         end if
      end if
   end repeat
   --- replace tables as arrays
   repeat with i = 1 to number of lines of tLua
      put line i of tLua into L
      set itemDel to "("
      if item 1 of L = "table.insert" then 
         put   item 2 of L into tArrayPart
         put item 3 to -1 of tArrayPart into tValues
         set itemDel to comma
         put word 1  of item 2 of L into tArray
         
         replace "{" with empty in tValues
         replace "}" with empty in tValues
         replace ")" with empty in tValues
         put "put" && tValues &&"into" && tArray into tNewLine
         put tNewLine into line 1 of tLua
      end if
   end repeat
   replace "#" with space & "the number of lines of the keys of" & space in tLua
   replace ", into" with " into" in tLua
   replace "==" with "=" in tLua
   put tLua
   --put tLua into fld 1
end translate.luaToaDegree

      set the rect of the templateImage to rect of image 1
      set the imageData of the templateImage to tImageData
   end if
   
   -- shrink the image, larger image = VERY LONG WAIT
   set the rect of the templateImage to 0,0,pGridWidth,pGridHeight
   create image pImage
   set itemDel to comma
   put the width of image pImage into tImageWidth
   put the height of image pImage into tImageHeight
   put the imageData of image pImage into pImageData
   put empty into gBinaryStrings
   --set the custompropertyset of stack (The mainStack of this stack) to Fontdata
   --put "ImageImport" into gCurrentFont
   put 0 into gCurrentAsciiVal

   put 0 into startY

   set the cursor to busy

   --repeat with startX = 1 to tImageWidth step gGridHeight
   -- all we need is this string
   put getImageBinaryColor(pGridWidth, pGridHeight, pImage, startX, startY) into tBinaryString
   
   --- bitwise is not needed for this, it's for bitfontmaker2 data

function getImageBinaryColor pWidth,pHeight,pImage
   set itemDel to comma
   repeat with y = 1 to pHeight
      put "" into tBitmapLine -- Initialize the line string
     
         repeat with x =  1 to pWidth
            put getPixelRGBA("16x16",x,y) into tRGBA  --pimage was 
            -- if there is no alpha then item 4 would be 255 on every line
            put addallLines(4,tRGBA) into allrgba
            if addallLines(4,tRGBA) < (the number of lines of tRGBA)*255 then
               -- imports transparency
               if tRGBA is "0,0,0,0" then
                  put "0"  after tBitmapLine -- White pixel
               else
                  put "1" after tBitmapLine -- Non-white pixel
               end if
            else
               -- imports flat image
               if tRGBA is "255,255,255,255" then
                  put "0"  after tBitmapLine -- White pixel
               else
                  put "1" after tBitmapLine -- Non-white pixel
               end if
            end if
         end repeat
         put tBitmapLine & cr after tBitmap
      end repeat
      --put allRGBA
      return tBitmap
   end getImageBinaryColor

function getPixelRGBA pImg,x,y -- return R,G,B,A
   put the width of img pImg into w
   put 4*((y-1)*w+x) into k
   put byte k-2 to k of the imageData of img pImg into b
   put byteToNum(byte (y-1)*w+x of the alphaData of img pImg) into c
   return (byteToNum(byte 1 of b),byteToNum(byte 2 of b), byteToNum(byte 3 of b), c) 
end getPixelRGBA


-- bitwise operation turns long 010101 style binary to compressed bitwise binary for BitFontMaker2 (and cleaner data)
Function BuildBitWise gBinaryString
   put "1,2,4,8,16,32,64,128,256,512,1024,2048,4096,8192,16384,32768" into columnValues
   put gBinaryString into tBits
   set itemDel to comma
   repeat with i = 1 to gGridColumns
      put 0 into tLineTotal
      repeat with j = 1 to gGridRows
         if char j of line i of tBits is not 0 then add item j of columnValues to  tLineTotal
      end repeat
      put tLineTotal & comma after AllTotals
   end repeat
   --put AllTotals
   return char 1 to -2 of AllTotals
end BuildBitWise

function getFilledColumns inputValue
   put "1,2,4,8,16,32,64,128,256,512,1024,2048,4096,8192,16384,32768" into columnValues
   set itemDel to comma
   put empty into filledColumns
   repeat with i = number of items in columnValues down to 1
      if inputValue >= item i of columnValues then
         put i & comma after filledColumns
         
         subtract item i of columnValues from inputValue
      end if
   end repeat
   delete the last char of filledColumns -- remove trailing comma
   return filledColumns
end getFilledColumns

--// rotate the points of an object
function Math.rotatePoints x, y, cx, cy, tAngle 
   --// x,     //X coords to rotate - replaced on return
   --// y,     //Y coords to rotate - replaced on return
   --// cx,      //X coordinate of center of rotation
   --// cy,      //Y coordinate of center of rotation
   -- // angle)   //Angle of rotation (radians, counterclockwise)
   put cos(tAngle) into tcos 
   put sin(tAngle) into tsin  
   put ((x-cx)*tcos - (y-cy)*tsin) + cx into Xout 
   put ((x-cx)*tsin + (y-cy)*tcos) + cy into Yout
   return xOut & comma & yOut
end Math.rotatePoints


function Math.clamp pValue, pMin, pMax
   if pValue < pMin then
      return pMin
   else if pValue > pMax then
      return pMax
   else
      return pValue
   end if
end Math.clamp

function Math.rectAtPoint x,y,w,h
   if w <1 then put 1 into w
   if h < 1 then put 1 into h
   if x < 1 then put 1 into x
   if y < 1 then put 1 into y
   return x-(w/2),y-(h/2),x+(w/2),y+(h/2)
end Math.rectAtPoint

function Math.scaleRect tPercentage,trect
   put tPercentage*0.01 into tPercentage
   set itemDel to comma
   put item 1 of tRect into x1
   put item 2 of tRect into y1
   put item 3 of tRect into x2
   put item 4 of trect into y2
   
   put (x2-x1)*tPercentage into tWidth
   put (y2-y1)*tPercentage into tHeight
   
   put x1+(x2-x1)/2 into xcenter
   put y1+(y2-y1)/2 into yCenter
   
   put  xCenter - (tWidth/2) into x1
   put yCenter - (tHeight/2) into y1
   put  xCenter + (tWidth/2) into x2
   put yCenter +(tHeight/2) into y2
   return x1,y1,x2,y2
end Math.scaleRect

--- converts n lines of repeating vectors to vx,vy,count
function ProcessRepeatingVectors tData
   put "" into tNewData
   put "" into tLastLine
   filter lines of tData without empty
   Replace "-10" with "-1" in tData
   Replace "10" with "1" in tData
   repeat for each line L in tData
      if L is  tLastLine then
         put L into tLastLine
         put L & cr after tRepeatList
         put number of lines of tRepeatList into tRepeatCount
      else
         put tLastLine & comma & tRepeatCount & cr after tVectorIterationData
         put L into tLastLine
         put 1 into tRepeatCount
         put "" into tRepeatList
      end if
   end repeat
   filter lines of tVectorIterationData without empty
   return line 2 to -1 of tVectorIterationData
end ProcessRepeatingVectors


function Prop.ProcessATrailToVectors tDataSet,tProp,tKey
   set the customPropertySet of this card to tDataSet 
   put the TrailArray of this card  into tArray
   set the customPropertySet of this card to empty
   put the keys of tArray into KeyLines
   sort lines of keyLines numeric ascending
   filter lines of KeyLines without empty
   if tKey is a number then put line tKey of KeyLines into tKey
   put tArray[tKey] into tPoints
   filter lines of tPoints without empty
   put the number of lines of tPoints into N
   repeat with i = N to 2 step -1
      --- subtract start point from end  point 
      put item 1 of line i of tPoints - item 1 of line i-1 of tPoints into vectorX
      put item 2 of line i of tPoints - item 2 of line i-1 of tPoints into vectorY
      if vectorX is  0 and vectorY is  0  then next repeat
      put vectorX & comma & vectorY & cr before newPoints
   end repeat
   put ProcessRepeatingVectors(newPoints) into tVectorIncrementData
   return tVectorIncrementData
end Prop.ProcessATrailToVectors

on doTrace pImageName, pGraphicName
   put the milliseconds into tTimeItTakes
   -- load the maskData of a png with transparency
   -- could be adjusted to alphaData of a png with transparency
   -- if you load a jpg/png without transparency you will get a border of the image boundaries of the dimensions of the image
   
   put the maskData of image pImageName into tAlpha
   if tAlpha = "" then exit doTrace
   
   put the width of image pImageName into tWidth
   put the height of image pImageName into tHeight
   
   set the cursor to busy
   
   put the left of image pImageName -1 into tLeft  -- -1 compensates for the 1 pixel border of the graphic
   put the top of image pImageName  - 1 into tTop -- -1 compensates for the 1 pixel border of the graphic
   
   put "" into tAlphaArray
   
   -- get custom property that looks up where to start sweep depending on angle
   -- see at bottom of script what is in uRadialSweepAngle
   put the uRadialSweepAngle of this card into tSweepArray
   
   put the number of lines of the keys of tSweepArray into sSweepExtent
   
   --   put the maskData of the image into an array
   --   with row number as index (=y value) and x values as comma delimited
   --   list as values
   --   all values relative to the image, not the card
   
   put the milliseconds into tStart
   repeat with i = 0 to tHeight -1
      repeat with j = 1 to tWidth
         put (charToNum(char tWidth * i + j of tAlpha)) into  tValue
         put tValue into tAlphaArray[i+1][j]
      end repeat
   end repeat
   
   put the keys of tAlphaArray into tKeys
   sort tKeys numeric
   
   put 0 into tLastItemValue -- tLastItemValue is used to find transitions e.g. from transparent to non-transparent
   
   -- the big repeat iterating over evey pixel
   repeat for each line aKey in tKeys -- aKey is the row number
      
      repeat with i = 1 to tWidth -- iterate over each item in tRow
         
         put tAlphaArray[aKey][i] into tItemValue
         if tItemValue > kThreshold and tItemValue <> tLastItemValue then  -- criterion met AND border 
            put i  & "," & aKey  into startLoc
            put tItemValue into tLastItemValue
         else
            put tItemValue into tLastItemValue
            next repeat 
         end if
         if (tFoundLocsArray[startLoc]) then next repeat -- it was already traced
         
         -- now we have found a condition to start tracing 
         
         -- tTerminateMinusOne is part of the termination rule
         -- put first and second location of a trace and terminate if search finds the same sequence again = End of a trace
         put startLoc into tTerminateMinusOne
         put startLoc & return after tLinePoints
         
         -- tFoundLocsArray keeps track of all points that were found in a shape
         put true into tFoundLocsArray[startLoc]
         
         put 180 into sStartSweepAt -- used by function radialSweep as starting point,  serves as memory of where to start sweep
         
         put item 1 of startLoc into x
         put item 2 of startLoc into y
         
         -- this is first test on newly found contour pixel to test and set up the contour tracing
         put radialSweep(x, y, tSweepArray, tAlphaArray) into tRes
         if tRes = "nada" then 
            -- "nada" is returned if no neighbor is found
            
            -- if single points are to be excluded activate this and block further down
            -- delete last line of tLinePoints
            
            -- this "closes" single points and records them, if you don't want this block this line
            put startLoc & return & return after tLinePoints
            put true into tFoundLocsArray[startLoc]
            
            put kThreshold into tLastItemValue
            
            next repeat -- repeat with i = 1 to tWidth
         end if
         
         -- we found begin of a trace
         put tRes into startLoc
         
         put tRes & return after tLinePoints
         put true into tFoundLocsArray[startLoc]
         
         put tRes into tTerminate
         put "" into tBeforeLastLoc
         
         -- now we follow this new trace = one contour
         repeat  -- iterate one contour
            put item 1 of startLoc into x
            put item 2 of startLoc into y
            put radialSweep(x, y, tSweepArray, tAlphaArray) into tRes
            if tRes = "nada" then 
               put kThreshold into  tLastItemValue
               -- no bordering pixel found -> exit repeat
               exit repeat -- iterate one contour
            end if
            
            put tRes into tLastLoc
            put tRes & return after tLinePoints
            put true into tFoundLocsArray[tRes]
            
            -- the termination algorithm, it compares if first and second pixel of current contour have been visited and if so exits
            if tLastLoc = tTerminate and tBeforeLastLoc = tTerminateMinusOne then                
               delete last line of tLInePoints -- part of termination algo, not needed.
               put cr  after tLinePoints
               exit repeat -- iterate one contour
            end if
            
            put tRes into startLoc
            put tLastLoc into tBeforeLastLoc
         end repeat -- iterate one contour
         
      end repeat -- iterate content of row = column = with i = 1 to tWidth
      
      if aKey mod 40 = 0 then set cursor to busy -- give feedback
      put kThreshold into tLastItemValue -- we will start at a new row
   end repeat -- iterate number of row = height 
   
   put "" into tCollect
   -- adjust image coordinates to card card coordinates
   repeat for each line aLine in tLinePoints
      if aLine = empty then 
         put cr after tCollect
         next repeat
      end if
      add tLeft to item 1 of aLine
      add tTop to item 2 of aLine
      put aLine & cr after tCollect
   end repeat
   
   delete last char of tCollect -- a return
   
   put tCollect into tLinePoints
   
   lock screen
   set the points of grc pGraphicName to tLinePoints
   
   set the originalWidth of graphic pGraphicName to width of graphic pGraphicName
   set the originalHeight of graphic pGraphicName to width of graphic pGraphicName 
   Prop.appendTraceHistory tLinePoints
   --Put number of lines of tLinePoints into fld "NumPoints"
   unlock screen
   --put "that took:" && the milliseconds-tTimeItTakes && "Ms"
end doTrace

private function radialSweep @px, @py, @pSweepArray, @pAlphaArray
   
   put  sStartSweepAt into i
   
   repeat sSweepExtent 
      put px + item 1 of pSweepArray[i] into tNewX
      put py + item 2 of pSweepArray[i] into tNewY
      put  pAlphaArray[tNewY][tNewX] into tTest
      if tTest > kThreshold then
         
         put tNewX - px into dX
         put tNewY - py into dY
         
         --  tAngleFromWhere  contains the angle in degrees from where we came to new tNewX,tNewY
         -- i.e. if angle is 45 last x,y was at northEast, if angle is 180 last x,y was at West
         put (round(abs (atan2(dY,dX)*180/pi-180)))  into  tAngleFromWhere  
         put tAngleFromWhere into sStartSweepAt
         return tNewX,tNewY
      end if
      
      -- we go clockwise, hence subtract
      subtract 45 from i
      add 360 to i
      put i mod 360 into i
      
   end repeat
   return "nada"
end radialSweep



-- the content of the array uRadialSweepAngle (a custom property) of this card
-- used to set the starting of the sweep depending on the angle of the previous pixel
--270 -1,+1
--225 -1,+0
--180 -1,-1
--135 -0,-1
--90 +1,-1
--45 +1,+0
--0 +1,+1
--315 +0,+1

--- use these to transfer the Properties
on setProperty.uRadialSweep
   put -1,+1 into tArray[1]
   put -1,+0 into tArray[2]
   put -1,-1 into tArray[3]
   put -0,-1 into tArray[4]
   put +1,-1 into tArray[5]
   put +1,+0 into tArray[6]
   put +1,+1 into tArray[7]
   put +0,+1 into tArray[8]
   set the uRadialSweep of this card to tArray
end setProperty.uRadialSweep

on setProperty.uRadialSweepAngle
   put +1,+0 into tArray[45]
   put -0,-1 into tArray[135]
   put +1,+1 into tArray[0]
   put -1,+0 into tArray[225]
   put -1,+1 into tArray[270]
   put -1,-1 into tArray[180]
   put +1,-1 into tArray[90]
   put +0,+1 into tArray[315]
   set the uRadialSweepAngle of this card to tArray
end setProperty.uRadialSweepAngle

on getProperty.uRadialSweepAngle
   put the uRadialSweepAngle of this card into tArray
   put the keys of tArray into tKeys
   repeat for each line K in tKeys
      put "put" && tArray[K] && "into tArray["&K&"]" & cr after tProperSetUp
   end repeat
   put tProperSetUp
end getProperty.uRadialSweepAngle
